cmake_minimum_required(VERSION 3.0.2)
project(drone_slam_utils)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_BUILD_TYPE Release)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs livox_ros_driver)
catkin_package()
include_directories(${catkin_INCLUDE_DIRS})
add_executable(livox_to_pointcloud2 src/livox_to_pointcloud2.cpp)
add_dependencies(livox_to_pointcloud2 ${catkin_EXPORTED_TARGETS})
target_link_libraries(livox_to_pointcloud2 ${catkin_LIBRARIES})
install(TARGETS livox_to_pointcloud2 RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
